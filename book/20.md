## 20.1　入力・モデル生成・解法・出力を分離する

**マリゴールド**: 第1章の 1.5 で見た流れを、ファイルの分け方にそのまま当てはめよう。

## 20.2　データとモデルを独立させる

**マリゴールド**: 入力の部品は、第17章の関数をまとめたものだよ。1つ違うのは、問題が見つかったら **例外** を投げることだ。

`factory_opt/data_io.py`:

```python
"""入力：CSV を読み、検証し、モデル用の辞書にする"""
from pathlib import Path

import pandas as pd

FILES = {
    "products": ["product", "label", "price", "material", "demand", "min"],
    "lines": ["line", "label", "cost_per_h", "hours"],
    "line_time": ["product", "line", "hours"],
    "resources": ["resource", "label", "capacity"],
    "usage": ["product", "resource", "amount"],
}


class DataError(Exception):
    """入力データに問題があるときに投げる例外"""


def load_data(folder: Path) -> dict:
    return {name: pd.read_csv(Path(folder) / f"{name}.csv") for name in FILES}


def validate(data: dict) -> list[str]:
    problems = []
    for name, cols in FILES.items():
        df = data[name]
        missing = [c for c in cols if c not in df.columns]
        if missing:
            problems.append(f"{name}.csv: 列がない {missing}")
            continue
        blank = df[df[cols].isna().any(axis=1)].index + 2
        if len(blank):
            problems.append(f"{name}.csv: 空欄がある（{blank.tolist()} 行目）")
        numeric = df[cols].select_dtypes("number")
        negative = df[(numeric < 0).any(axis=1)].index + 2
        if len(negative):
            problems.append(f"{name}.csv: マイナスの値がある（{negative.tolist()} 行目）")
    if problems:
        return problems

    products = set(data["products"]["product"])
    lines = set(data["lines"]["line"])
    resources = set(data["resources"]["resource"])
    for key, df in [("product", data["products"]), ("line", data["lines"]), ("resource", data["resources"])]:
        dup = df[df[key].duplicated()][key].tolist()
        if dup:
            problems.append(f"{key} が重複している: {dup}")
    for p in set(data["line_time"]["product"]) - products:
        problems.append(f"line_time.csv: 知らない商品 '{p}'")
    for l in set(data["line_time"]["line"]) - lines:
        problems.append(f"line_time.csv: 知らないライン '{l}'")
    for p in set(data["usage"]["product"]) - products:
        problems.append(f"usage.csv: 知らない商品 '{p}'")
    for r in set(data["usage"]["resource"]) - resources:
        problems.append(f"usage.csv: 知らない資源 '{r}'")
    for p in products - set(data["line_time"]["product"]):
        problems.append(f"商品 '{p}' を作れるラインがない")
    bad = data["products"][data["products"]["min"] > data["products"]["demand"]]
    for p in bad["product"]:
        problems.append(f"商品 '{p}' の最低生産数が需要より大きい")
    return problems


def to_dicts(data: dict) -> dict:
    return {
        "products": data["products"].set_index("product").to_dict("index"),
        "lines": data["lines"].set_index("line").to_dict("index"),
        "line_time": data["line_time"].set_index(["product", "line"])["hours"].to_dict(),
        "capacity": data["resources"].set_index("resource")["capacity"].to_dict(),
        "usage": data["usage"].set_index(["product", "resource"])["amount"].to_dict(),
    }


def read_input(folder: Path) -> dict:
    """読み込み → 検証 → 辞書。問題があれば DataError"""
    data = load_data(folder)
    problems = validate(data)
    if problems:
        raise DataError("\n".join(problems))
    return to_dicts(data)
```

**初花**: 「知らないキー」の検査が、第17章の `iterrows` から、集合の引き算（`set(...) - products`）に変わってる。第17章の one more thing の話だね。

**マリゴールド**: よく気づいたね。データが大きくなっても速く動くようにしておいたよ。

モデルの部品は、辞書を受け取って PuLP の問題を返すだけです。

`factory_opt/model.py`:

```python
"""モデル：辞書を受け取り、PuLP の問題を作る（ファイルもソルバーも知らない）"""
import pulp


def unit_profit(d: dict, p: str, l: str) -> float:
    return d["products"][p]["price"] - d["products"][p]["material"] - d["line_time"][p, l] * d["lines"][l]["cost_per_h"]


def build_model(d: dict):
    products, lines, line_time = d["products"], d["lines"], d["line_time"]
    capacity, usage = d["capacity"], d["usage"]

    prob = pulp.LpProblem("factory", pulp.LpMaximize)
    make = {(p, l): prob.add_variable(f"make_{p}_{l}", lowBound=0, cat="Integer") for (p, l) in line_time}

    prob += pulp.lpSum(unit_profit(d, p, l) * make[p, l] for (p, l) in line_time), "total_profit"

    total = {p: pulp.lpSum(v for (q, l), v in make.items() if q == p) for p in products}
    for l in lines:
        prob += pulp.lpSum(line_time[p, m] * make[p, m] for (p, m) in line_time if m == l) \
            <= lines[l]["hours"], f"line_{l}"
    for r in capacity:
        prob += pulp.lpSum(usage.get((p, r), 0) * total[p] for p in products) <= capacity[r], f"res_{r}"
    for p in products:
        prob += total[p] <= products[p]["demand"], f"demand_{p}"
        prob += total[p] >= products[p]["min"], f"min_{p}"
    return prob, make
```

| 関数の引数と戻り値の型ヒント | 意味 |
|------------------------|------|
| `folder: Path` | 引数はフォルダのパス |
| `-> dict` | 辞書を返す |
| `-> list[str]` | 文字列のリストを返す |

型ヒントは Python の動作には影響しませんが、関数が何を受け取って何を返すかが一目で分かるので、部品に分けたコードでは特に役立ちます。

`factory_opt/solve.py`:

```python
"""解法：設定に従ってソルバーを選び、解いて、結果を普通の辞書で返す"""
import pulp

SOLVERS = {
    "HiGHS": pulp.HiGHS,
    "CBC": pulp.COIN_CMD,
    "GLPK": pulp.PYGLPK,
    "SCIP": pulp.SCIP_PY,
}


class SolveError(Exception):
    """使える解が得られなかったときに投げる例外"""


def make_solver(cfg: dict):
    name = cfg.get("name", "HiGHS")
    if name not in SOLVERS:
        raise SolveError(f"知らないソルバー: {name}")
    solver = SOLVERS[name](msg=cfg.get("msg", False), timeLimit=cfg.get("time_limit"), gapRel=cfg.get("gap_rel"))
    if not solver.available():
        raise SolveError(f"ソルバー {name} が使えません（インストールを確認してください）")
    return solver


def solve(prob, make, cfg: dict) -> dict:
    stats = prob.solve(make_solver(cfg))
    if not stats.has_solution:
        raise SolveError(f"解が得られませんでした: {stats.status_str}")
    return {
        "status": stats.status_str,
        "profit": pulp.value(prob.objective),
        "make": {k: v.value() for k, v in make.items()},
        "seconds": stats.time,
        "solver": stats.solver,
    }
```

**初花**: `solve` が返すのは、PuLP の変数じゃなくて、普通の数の辞書なんだね。

**マリゴールド**: そう。ここから先の検証と出力は、PuLP を知らなくていいようにしておくんだ。それと、第19章で見たように、`TimeLimit` でも解があれば使えるから、`Optimal` かどうかではなく `has_solution` で判断しているよ。

| 設定に出すもの | コードに残すもの |
|-------------|-------------|
| ソルバーの種類・時間制限・ギャップ | モデルの式 |
| 出力先のフォルダ | 検証のルール |
| （データのファイルの場所は引数で） | 表の作り方 |

## 20.4　最適化結果の自動検証

**マリゴールド**: 第6章で学んだ検証を、部品として独立させよう。モデルとは別に、**データだけで** 答えを確かめるんだ。

`factory_opt/verify.py`:

```python
"""検証：モデルとは別に、データだけで答えを確かめる"""
TOL = 1e-6


def verify(d: dict, make: dict) -> list[str]:
    problems = []
    total = {p: 0.0 for p in d["products"]}
    for (p, l), x in make.items():
        if abs(x - round(x)) > TOL:
            problems.append(f"{p}×{l}: 整数ではない ({x})")
        if x < -TOL:
            problems.append(f"{p}×{l}: マイナス ({x})")
        total[p] += x
    for l, info in d["lines"].items():
        used = sum(d["line_time"][p, m] * x for (p, m), x in make.items() if m == l)
        if used > info["hours"] + TOL:
            problems.append(f"ライン {l}: {used} 時間 > {info['hours']} 時間")
    for r, cap in d["capacity"].items():
        used = sum(d["usage"].get((p, r), 0) * x for p, x in total.items())
        if used > cap + TOL:
            problems.append(f"資源 {r}: {used} > {cap}")
    for p, info in d["products"].items():
        if total[p] > info["demand"] + TOL:
            problems.append(f"{p}: 需要 {info['demand']} を超える ({total[p]})")
        if total[p] < info["min"] - TOL:
            problems.append(f"{p}: 最低生産数 {info['min']} に届かない ({total[p]})")
    return problems
```

出力の部品は、結果を表にして保存します。

`factory_opt/report.py`:

```python
"""出力：結果を表にして、CSV と Excel に書き出す"""
from pathlib import Path

import pandas as pd


def to_tables(d: dict, result: dict) -> dict:
    plan = pd.DataFrame([
        {"product": p, "label": d["products"][p]["label"], "line": l, "cases": round(x)}
        for (p, l), x in result["make"].items() if round(x) > 0
    ])
    total = plan.groupby("product")["cases"].sum()
    resources = pd.DataFrame([
        {"resource": r, "capacity": cap,
         "used": sum(d["usage"].get((p, r), 0) * int(total.get(p, 0)) for p in d["products"])}
        for r, cap in d["capacity"].items()
    ])
    resources["rate"] = resources["used"] / resources["capacity"]
    summary = pd.DataFrame([{
        "profit": round(result["profit"]), "status": result["status"],
        "solver": result["solver"], "seconds": round(result["seconds"], 3),
    }])
    return {"summary": summary, "plan": plan, "resources": resources}


def save(tables: dict, folder: Path) -> list[Path]:
    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    paths = []
    for name, df in tables.items():
        path = folder / f"{name}.csv"
        df.to_csv(path, index=False, encoding="utf-8-sig")
        paths.append(path)
    xlsx = folder / "report.xlsx"
    with pd.ExcelWriter(xlsx) as writer:
        for name, df in tables.items():
            df.to_excel(writer, sheet_name=name, index=False)
    paths.append(xlsx)
    return paths
```

**初花**: 部品に分けたのはいいけど、来年だれかが `model.py` を直したとき、ちゃんと動いてるか確かめるのは大変じゃない？


テストに書くことは、ここまでの章で学んだ確かめ方そのものです。

| テスト | 確かめること | 学んだ章 |
|-------|-----------|--------|
| `test_base_data_gives_known_profit` | いつものデータで、いつもの答え（470,300円） | 第9・17章 |
| `test_small_data_matches_brute_force` | 小さなデータで、全探索と同じ答え | 第7章 |

`tests/test_factory.py`:

```python
import copy
import itertools
from pathlib import Path

import pytest

from factory_opt.data_io import DataError, load_data, read_input, validate
from factory_opt.model import build_model, unit_profit
from factory_opt.solve import SolveError, solve
from factory_opt.verify import verify

DATA = Path(__file__).parent.parent / "data"
CFG = {"name": "HiGHS", "msg": False, "time_limit": 60}

# 手で確かめられるくらい小さなデータ
SMALL = {
    "products": {
        "a": {"label": "A", "price": 100, "material": 40, "demand": 5, "min": 0},
        "b": {"label": "B", "price": 120, "material": 50, "demand": 4, "min": 1},
    },
    "lines": {"L": {"label": "ライン", "cost_per_h": 10, "hours": 6}},
    "line_time": {("a", "L"): 1.0, ("b", "L"): 1.5},
    "capacity": {"r": 10},
    "usage": {("a", "r"): 2, ("b", "r"): 1},
}


def brute_force(d):
    keys = list(d["line_time"])
    best = None
    for counts in itertools.product(range(11), repeat=len(keys)):
        make = dict(zip(keys, counts))
        if not verify(d, make):   # 検証で問題がなければ、条件を満たす案
            profit = sum(unit_profit(d, p, l) * x for (p, l), x in make.items())
            best = profit if best is None else max(best, profit)
    return best


def test_base_data_gives_known_profit():
    """いつものデータで、いつもの答え（第9章・第17章）が出る"""
    d = read_input(DATA)
    prob, make = build_model(d)
    result = solve(prob, make, CFG)
    assert result["status"] == "Optimal"
    assert result["profit"] == pytest.approx(470300)
    assert verify(d, result["make"]) == []


def test_small_data_matches_brute_force():
    """小さなデータでは、全探索と同じ答えになる（第7章）"""
    prob, make = build_model(SMALL)
    result = solve(prob, make, CFG)
    assert result["profit"] == pytest.approx(brute_force(SMALL))


def test_validate_finds_unknown_product():
    data = load_data(DATA)
    data["usage"].loc[len(data["usage"])] = ["dresing", "ume_kg", 2]
    assert "usage.csv: 知らない商品 'dresing'" in validate(data)


def test_read_input_raises_on_bad_data(tmp_path):
    data = load_data(DATA)
    data["products"].loc[1, "material"] = None
    for name, df in data.items():
        df.to_csv(tmp_path / f"{name}.csv", index=False)
    with pytest.raises(DataError, match="空欄"):
        read_input(tmp_path)


def test_impossible_minimum_raises():
    """最低生産数がラインの能力を超えると、解がないことを例外で知らせる"""
    d = copy.deepcopy(SMALL)
    d["products"]["b"]["min"] = 4
    d["products"]["a"]["min"] = 5
    prob, make = build_model(d)
    with pytest.raises(SolveError):
        solve(prob, make, CFG)


def test_verify_catches_overuse():
    bad = {("a", "L"): 5, ("b", "L"): 2}   # ライン 5 + 3 = 8 時間 > 6 時間
    assert any("ライン L" in msg for msg in verify(SMALL, bad))
```

**初花**: 6個とも PASSED！ 全探索の関数で、`verify` を「条件を満たしているか」の判定に使い回してるのが賢い。

**マリゴールド**: 部品が独立していると、テストでも組み合わせて使えるんだ。`tmp_path` は pytest が用意してくれる一時フォルダで、テストのためのファイルをそこに作れるよ。

| pytest の道具 | 使い方 |
|-------------|-------|
| `assert 条件` | 条件が偽ならテストは失敗 |
| `pytest.approx(値)` | 小数の誤差を許して比べる |
| `pytest.raises(例外)` | その例外が起きることを確かめる |
| `tmp_path` | テスト用の一時フォルダ |

**初花**: 再現性っていうのは、同じデータなら、いつ誰が実行しても同じ答えになる、ってこと？

**マリゴールド**: そう。そのためには、データだけでなく **道具のバージョン** も記録しておく必要がある。第2章で作った `requirements.txt` にバージョンを固定して書いておくこと、それに実行のたびにバージョンをログに残すことが大事だよ。

| 再現性のために残すもの | 方法 |
|-------------------|------|
| ライブラリのバージョン | `requirements.txt` に `pulp[highs]==4.0.1` のように固定 |
| 実行時のバージョン | ログの最初に PuLP・Python・ソルバーを出す（20.6） |
| 乱数の種 | 第12章の `random.Random(seed)` のように固定する |
| モデルそのもの | 必要なら `writeMPS` で保存する（第19章） |

⚠️ 第6章や第15章で見たように、最適解が複数あるときは、ソルバーやバージョンによって選ばれる答えが変わることがあります。テストでは、変数の値そのものより、**目的関数の値** と **検証の結果** を確かめるほうが安定します。

## 20.6　実行ログとエラー処理

**マリゴールド**: 最後は、全体の流れをまとめる `main.py`。毎週自動で動かすなら、**何が起きたかを記録するログ** と、**失敗したときに分かりやすく止まるエラー処理** が欠かせない。

`factory_opt/main.py`:

```python
"""実行の入口：python -m factory_opt.main --data data --config config.toml"""
import argparse
import logging
import platform
import sys
import tomllib
from pathlib import Path

import pulp

from .data_io import DataError, read_input
from .model import build_model
from .report import save, to_tables
from .solve import SolveError, solve
from .verify import verify

log = logging.getLogger("factory_opt")


def run(data_folder: Path, config: dict) -> dict:
    log.info("PuLP %s / Python %s / ソルバー %s", pulp.__version__, platform.python_version(), config["solver"]["name"])
    log.info("入力を読み込みます: %s", data_folder)
    d = read_input(data_folder)
    log.info("商品 %d 件・ライン %d 本・資源 %d 種類", len(d["products"]), len(d["lines"]), len(d["capacity"]))

    prob, make = build_model(d)
    log.info("モデル: 変数 %d 個・制約 %d 本", prob.numVariables(), prob.numConstraints())

    result = solve(prob, make, config["solver"])
    log.info("解きました: %s / 利益 %s 円 / %.3f 秒", result["status"], f"{result['profit']:,.0f}", result["seconds"])

    problems = verify(d, result["make"])
    if problems:
        raise SolveError("検証に失敗しました:\n" + "\n".join(problems))
    log.info("検証: OK")

    paths = save(to_tables(d, result), Path(config["output"]["folder"]))
    log.info("保存しました: %s", ", ".join(p.name for p in paths))
    return result


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="うめ食品の生産計画を作る")
    parser.add_argument("--data", type=Path, default=Path("data"))
    parser.add_argument("--config", type=Path, default=Path("config.toml"))
    args = parser.parse_args(argv)

    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s",
                        handlers=[logging.StreamHandler(), logging.FileHandler("run.log", encoding="utf-8")])
    with open(args.config, "rb") as f:
        config = tomllib.load(f)

    try:
        run(args.data, config)
    except DataError as e:
        log.error("入力データに問題があります:\n%s", e)
        return 1
    except SolveError as e:
        log.error("%s", e)
        return 2
    return 0


if __name__ == "__main__":
    sys.exit(main())
```


**初花**: 最後の数字は何？

**マリゴールド**: プログラムの **終了コード** だよ。0なら成功、1なら入力データの問題、2なら解けなかった・ソルバーの問題、と決めておいた。毎週自動で動かす仕組み（タスクスケジューラや cron など）は、この数字を見て「失敗したらメールを送る」といったことができるんだ。

| 例外 | 起きるとき | 終了コード | ログ |
|------|----------|---------:|------|
| （なし） | 成功 | 0 | INFO のみ |
| `DataError` | 入力データの問題 | 1 | 何行目の何が悪いか |
| `SolveError` | 解がない・ソルバーが使えない・検証の失敗 | 2 | 理由 |

| ログのレベル | 使いどころ |
|------------|----------|
| `INFO` | 正常な流れの記録（読み込んだ件数、利益など） |
| `WARNING` | 続けられるが注意が必要（時間制限で止まった、など） |
| `ERROR` | 続けられない失敗 |

⚠️ ログに個人情報（スタッフの名前など）を書くときは、ログファイルの保管場所と閲覧できる人に気をつけましょう。

## 初花のメモ帳

- 入力・モデル・解法・検証・出力を別のファイルに分け、それぞれが知ることを少なくする
- 変えたい設定は TOML に出し、問題は例外（`DataError`・`SolveError`）で知らせ、終了コードで結果を返す
- テストには「いつもの答え」「小さなデータでの全探索」「データの間違い」「解けないとき」を書く。値より目的関数と検証の結果を確かめる

## マリゴールドの one more thing

**テストを自動で動かすと、壊れたことにすぐ気づける。**

GitHub などでコードを管理している場合、変更を送るたびに `python -m pytest` を自動で実行する仕組み（CI：継続的インテグレーション）を設定できます。GitHub Actions なら、`.github/workflows/test.yml` に「Python を用意して、`pip install -r requirements.txt` して、`pytest` を実行する」と数行書くだけです。だれかがモデルを直してテストが失敗すれば、取り込む前に気づけます。最適化のプログラムは、答えがそれらしく見えても間違っていることがある（第7章）ので、人の目だけに頼らない仕組みが特に役立ちます。
